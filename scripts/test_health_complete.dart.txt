#!/usr/bin/env dart

import 'dart:convert';
import 'dart:io';
import 'package:dio/dio.dart';

/// Test completo para verificar que HealthResponse funciona con datos reales

void main() async {
  print('🧪 Testing HealthResponse with real MCP Server data...\n');
  
  final dio = Dio();
  const serverUrl = 'http://192.168.1.6:10600/health';
  
  try {
    // Obtener datos reales del servidor
    print('1️⃣ Fetching real data from MCP Server...');
    final response = await dio.get(serverUrl);
    
    print('✅ Server responded successfully');
    print('   Raw response: ${response.data}\n');
    
    // Simular el parsing que hace Flutter
    print('2️⃣ Testing HealthResponse.fromJson...');
    
    // Aquí simularíamos la clase HealthResponse real
    final json = response.data as Map<String, dynamic>;
    
    print('✅ JSON parsing successful');
    print('   Status: ${json['status']}');
    print('   GCT Connected: ${json['gct_connected']}');
    print('   Tools Count: ${json['tools_count']}');
    print('   Uptime: ${json['uptime']} (raw)');
    print('   Error Count: ${json['error_count']}');
    print('   Request Count: ${json['request_count']}');
    print('   Sandbox: ${json['sandbox']}\n');
    
    // Test uptime parsing
    print('3️⃣ Testing uptime parsing...');
    final uptimeRaw = json['uptime'];
    final uptimeParsed = _parseUptime(uptimeRaw);
    print('   Raw uptime: "$uptimeRaw"');
    print('   Parsed uptime: $uptimeParsed seconds');
   // print('   Formatted: ${_formatUptime(uptimeParsed)}\n');
    
    print('🎉 All tests passed! HealthResponse should work correctly now.');
    
  } catch (e) {
    print('❌ Test failed: $e');
    if (e is DioException && e.response != null) {
      print('   Status: ${e.response!.statusCode}');
      print('   Response: ${e.response!.data}');
    }
    exit(1);
  }
}
